# D05 饼图配标签

3 到 5 个类别的占比，扇区内直接写类别名、不需精确读数时用，要精确比较改用柱图。改数据只改脚本里 D 的 names、values；值为 0 或缺失的类别自动不画。放不进扇区的名称只显示前 2 个字，读数里仍是全名。增减类别时同步改 D 与扇区颜色类。焦点是第一扇区，悬停或 Tab 聚焦时外移并显示读数。示例内容只是占位，使用时整套换成用户的内容，包括标题、副标题、标签、页脚和一句话概述。

```html
<div class="fg fx-d05" data-fig="d05">
<h2 class="fg-sr">饮品销量五类占比的饼图，扇区内标注类别名。</h2>
<div class="fg-panel">
<div class="fg-head"><div class="fg-title fx-c">饮品销量占比：上周</div></div>
<div class="fx-pie">
<svg class="fg-svg" viewBox="0 0 360 242" role="img" aria-label="饮品销量五类占比饼图，扇区内标注类别名，示例数据"></svg>
<div class="fx-tip"></div>
</div>
<div class="fg-foot">示例数据；扇区从 12 点方向顺时针排列，按销量从大到小。</div>
</div>
</div>
<script>
(function(){
var root=document.querySelector('.fg[data-fig="d05"]');
if(!root||root.getAttribute('data-ready'))return;
root.setAttribute('data-ready','1');
var D={unit:'杯',names:['咖啡','茶','果汁','牛奶','其他'],cls:['fx-s1','fx-s2','fx-s3','fx-s4','fx-s0'],values:[900,600,475,300,225]};
var NS='http://www.w3.org/2000/svg',total=0,W,CX,CY,R;
var wrap=root.querySelector('.fx-pie'),svg=root.querySelector('svg'),tip=root.querySelector('.fx-tip');
var S={secs:[],rings:[],labs:[],mid:[],cur:-1};
var items=[];
D.values.forEach(function(v,i){if(typeof v==='number'&&isFinite(v)&&v>0){items.push({i:i,v:v});total+=v}});
function el(tag,attrs,cls){var e=document.createElementNS(NS,tag);for(var k in attrs)e.setAttribute(k,attrs[k]);if(cls)e.setAttribute('class',cls);return e}
function pt(r,a){return (CX+r*Math.sin(a)).toFixed(2)+' '+(CY-r*Math.cos(a)).toFixed(2)}
function wedge(r,a0,a1){var big=a1-a0>Math.PI?1:0;return 'M'+CX+' '+CY+' L'+pt(r,a0)+' A'+r+' '+r+' 0 '+big+' 1 '+pt(r,a1)+' Z'}
function fits(x,y,hw,hh,a0,a1,m){
var t0x=Math.cos(a0),t0y=Math.sin(a0),t1x=Math.cos(a1),t1y=Math.sin(a1),ok=true;
[[x-hw,y-hh],[x+hw,y-hh],[x-hw,y+hh],[x+hw,y+hh]].forEach(function(p){
var qx=p[0]-CX,qy=p[1]-CY;
if(Math.sqrt(qx*qx+qy*qy)+m>R)ok=false;
if(qx*t0x+qy*t0y<m)ok=false;
if(-(qx*t1x+qy*t1y)<m)ok=false;
});
return ok;
}
function hide(){
if(S.cur>=0){
[S.secs[S.cur],S.labs[S.cur],S.rings[S.cur]].forEach(function(e){if(e)e.style.transform=''});
if(S.rings[S.cur])S.rings[S.cur].classList.remove('is-on');
S.cur=-1;
}
tip.classList.remove('is-on');
}
function show(i,focus){
hide();S.cur=i;
var m=S.mid[i],t='translate('+(4*Math.sin(m)).toFixed(2)+'px,'+(-4*Math.cos(m)).toFixed(2)+'px)';
S.secs[i].style.transform=t;
if(S.labs[i])S.labs[i].style.transform=t;
if(focus){S.rings[i].style.transform=t;S.rings[i].classList.add('is-on')}
tip.textContent=D.names[i]+' '+D.values[i]+' '+D.unit+' · '+(D.values[i]/total*100).toFixed(1)+'%';
tip.classList.add('is-on');
var sc=svg.getBoundingClientRect().width/W,tw=tip.offsetWidth,th=tip.offsetHeight;
var px=(CX+(R+14)*Math.sin(m))*sc,py=(CY-(R+14)*Math.cos(m))*sc;
tip.style.left=Math.max(0,Math.min(wrap.clientWidth-tw,px-tw/2))+'px';
tip.style.top=Math.max(0,Math.min(wrap.clientHeight-th,py-th/2))+'px';
}
function build(){
hide();
var box=svg.getBoundingClientRect().width||360,k=Math.min(360,box)/360;
W=360*k;CX=W/2;CY=121*k;R=117*k;
svg.setAttribute('viewBox','0 0 '+W+' '+(242*k));
[].slice.call(svg.querySelectorAll('.fx-gen')).forEach(function(e){e.parentNode.removeChild(e)});
S.secs=[];S.rings=[];S.labs=[];S.mid=[];
var a=0;
items.forEach(function(it){
var a1=a+it.v/total*2*Math.PI;
it.a0=a;it.a1=a1;S.mid[it.i]=(a+a1)/2;
var p=el('path',{d:wedge(R,a,a1),tabindex:'0','data-i':it.i},'fx-gen fx-sec '+D.cls[it.i]);
svg.appendChild(p);S.secs[it.i]=p;
a=a1;
});
items.forEach(function(it){
var m=S.mid[it.i],rl=R*0.57,x=CX+rl*Math.sin(m),y=CY-rl*Math.cos(m),nm=D.names[it.i],mg=k<1?6:0,ok;
var t=el('text',{x:x.toFixed(2),y:(y+4).toFixed(2),'text-anchor':'middle'},'fx-gen fg-label fx-lbl'+(it.i===0?' fx-on':''));
t.textContent=nm;
svg.appendChild(t);
var cands=nm.length>2?[nm,nm.slice(0,2)]:[nm];
for(var c=0;c<cands.length&&!ok;c++){t.textContent=cands[c];ok=fits(x,y,t.getComputedTextLength()/2,9,it.a0,it.a1,mg)}
if(ok)S.labs[it.i]=t;
else svg.removeChild(t);
});
items.forEach(function(it){
var g=el('path',{d:wedge(R+3,it.a0-3/R,it.a1+3/R)},'fx-gen fx-ring');
svg.appendChild(g);S.rings[it.i]=g;
});
}
wrap.addEventListener('mouseover',function(e){var i=e.target.getAttribute('data-i');if(i===null)hide();else show(+i,false)});
wrap.addEventListener('mouseleave',hide);
wrap.addEventListener('focusin',function(e){var i=e.target.getAttribute('data-i');if(i!==null)show(+i,true)});
wrap.addEventListener('focusout',hide);
build();
window.addEventListener('resize',build);
})();
</script>
```

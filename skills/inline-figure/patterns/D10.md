# D10 桑基流向

三列加权流向：带宽与次数成正比，焦点流为价格页到注册；悬停或聚焦一条流带或节点，其余降低。几何已按数据预先算好：增减节点或流带时，要重算节点高（次数×0.048）、同列间距 12、各带端点和落地页胶囊位置，并同步改悬停文字、摘要条、三段列表和页脚。示例内容只是占位，使用时整套换成用户的内容，包括标题、副标题、标签、页脚和一句话概述。

<script>
(function(){
var root=document.querySelector('.fg[data-fig="d10"]');
if(!root||root.getAttribute('data-ready'))return;
root.setAttribute('data-ready','1');
var bands=[].slice.call(root.querySelectorAll('.fx-band'));
var nodes=[].slice.call(root.querySelectorAll('[data-id]'));
var rects=[].slice.call(root.querySelectorAll('rect[data-id]'));
var rd=root.querySelector('.fx-read');
function ttl(e){return e.querySelector('title').textContent}
function show(bs,ids,msg){
bands.forEach(function(b){b.classList.toggle('is-off',bs.indexOf(b)<0)});
nodes.forEach(function(n){n.classList.toggle('is-off',ids.indexOf(n.getAttribute('data-id'))<0)});
if(msg!==undefined)rd.textContent=msg;
}
function reset(msg){
bands.forEach(function(b){b.classList.remove('is-off')});
nodes.forEach(function(n){n.classList.remove('is-off')});
if(msg!==undefined)rd.textContent=msg;
}
function flow(b,msg){show([b],[b.getAttribute('data-a'),b.getAttribute('data-b')],msg)}
function node(n,msg){
var id=n.getAttribute('data-id');
var bs=bands.filter(function(b){return b.getAttribute('data-a')===id||b.getAttribute('data-b')===id});
var ids=[id];
bs.forEach(function(b){ids.push(b.getAttribute('data-a'),b.getAttribute('data-b'))});
show(bs,ids,msg);
}
bands.forEach(function(b){
b.addEventListener('pointerenter',function(){flow(b)});
b.addEventListener('pointerleave',function(){reset()});
b.addEventListener('focus',function(){flow(b,ttl(b))});
b.addEventListener('blur',function(){reset('')});
});
rects.forEach(function(n){
n.addEventListener('pointerenter',function(){node(n)});
n.addEventListener('pointerleave',function(){reset()});
n.addEventListener('focus',function(){node(n,ttl(n))});
n.addEventListener('blur',function(){reset('')});
});
})();
</script>

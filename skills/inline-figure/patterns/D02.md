# D02 分组柱

两个并列序列要逐组比大小时用（3 到 12 个类目）。换数据改 D.cats 与 D.series（缺值写 null），换文案改标题、图例、类目名与页脚；增减类目要同步改 SVG 类目名与 D.cats。纵轴最大值由 `D.max` 决定，网格线与静态标签要同步改。焦点是工作日柱，悬停或 Tab 聚焦显示读数。示例内容只是占位，使用时整套换成用户的内容，包括标题、副标题、标签、页脚和一句话概述。

<script>
(function(){
var root=document.querySelector('.fg[data-fig="d02"]');
if(!root||root.getAttribute('data-ready'))return;
root.setAttribute('data-ready','1');
var D={unit:'千单',max:60,cats:['早高峰','午间','下午','晚高峰','夜间'],series:[{name:'工作日',cls:'fx-s1',values:[34,18,26,41,12]},{name:'周末',cls:'fx-s2',values:[22,27,31,29,16]}]};
var NS='http://www.w3.org/2000/svg',n=D.cats.length,x0=20,yT=16,yB=233,H=273;
var svg=root.querySelector('svg'),wrap=root.querySelector('.fx-wrap'),tip=root.querySelector('.fx-tip');
var grid=[].slice.call(svg.querySelectorAll('.fx-grid')),labs=[].slice.call(svg.querySelectorAll('.fg-t-dim'));
var S={x1:620,cw:120,cx:[],top:[],hits:[]};
function el(tag,attrs,cls){var e=document.createElementNS(NS,tag);for(var k in attrs)e.setAttribute(k,attrs[k]);if(cls)e.setAttribute('class',cls);return e}
function num(v){return v===null||v===undefined||v===''||!isFinite(Number(v))?null:Number(v)}
function Y(v){return yB-v/D.max*(yB-yT)}
function hide(){tip.classList.remove('is-on')}
function show(i){
hide();
var parts=[D.cats[i]];
D.series.forEach(function(s){var v=num(s.values[i]);if(v!==null)parts.push(s.name+' '+v+' '+D.unit)});
tip.textContent=parts.join(' · ');
tip.classList.add('is-on');
var tw=tip.getBoundingClientRect().width,th=tip.getBoundingClientRect().height,cx=S.cx[i],top=S.top[i]-th-8;
if(top<yT)top=S.top[i]+8;
tip.style.left=Math.max(x0,Math.min(S.x1-tw,cx-tw/2))+'px';
tip.style.top=top+'px';
}
function draw(){
hide();
var W=Math.min(640,svg.getBoundingClientRect().width||640);
S.x1=W-20;S.cw=(S.x1-x0)/n;
var bw=S.cw*.3,gap=8*(S.x1-x0)/600;
svg.setAttribute('viewBox','0 0 '+W+' '+H);
grid.forEach(function(g,k){var y=k===0?yT:k===1?(yT+yB)/2:yB;g.setAttribute('y1',y);g.setAttribute('y2',y);g.setAttribute('x2',S.x1)});
labs.forEach(function(t,i){t.setAttribute('x',x0+S.cw*(i+.5));t.setAttribute('y',yB+30)});
[].slice.call(svg.querySelectorAll('.fx-gen')).forEach(function(e){e.parentNode.removeChild(e)});
S.hits.forEach(function(h){wrap.removeChild(h)});
S.cx=[];S.top=[];S.hits=[];
for(var i=0;i<n;i++){
var cx=x0+S.cw*(i+.5),left=cx-(2*bw+gap)/2,tops=[];
S.cx.push(cx);
D.series.forEach(function(s,k){
var v=num(s.values[i]);if(v===null)return;
var y=Y(v);tops.push(y);
svg.appendChild(el('rect',{x:left+k*(bw+gap),y:y,width:bw,height:yB-y,rx:Math.min(8,bw/2)},'fx-gen '+s.cls));
});
S.top.push(tops.length?Math.min.apply(null,tops):yB);
var h=document.createElement('span');
h.className='fx-hit';h.tabIndex=0;h.setAttribute('data-i',i);
h.setAttribute('aria-label',D.cats[i]);
h.style.left=(x0+S.cw*i)+'px';h.style.width=S.cw+'px';h.style.height=H+'px';
wrap.appendChild(h);S.hits.push(h);
}
}
wrap.addEventListener('mouseover',function(e){var h=e.target.closest('.fx-hit');if(h)show(+h.getAttribute('data-i'))});
wrap.addEventListener('mouseleave',hide);
wrap.addEventListener('focusin',function(e){var i=e.target.getAttribute('data-i');if(i!==null)show(+i)});
wrap.addEventListener('focusout',hide);
draw();
window.addEventListener('resize',draw);
})();
</script>

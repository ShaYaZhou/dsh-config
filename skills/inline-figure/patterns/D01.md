# D01 折线趋势

同一指标随时间变化，看走势与拐点时用（4 到 12 个时间点，只画一条线）。换数据改 D.values（缺值写 null），换文案改标题、横轴标签、页脚与概述；增减时间点同步改横轴标签与 D.values。纵轴最大值由 `D.max` 决定，网格线与静态标签要同步改。焦点是整条线，悬停或 Tab 聚焦某点出读数。示例内容只是占位，使用时整套换成用户的内容，包括标题、副标题、标签、页脚和一句话概述。

<script>
(function(){
var root=document.querySelector('.fg[data-fig="d01"]');
if(!root||root.getAttribute('data-ready'))return;
root.setAttribute('data-ready','1');
var D={name:'活跃人数',unit:'千人',max:60,values:[42,47,51,46,39,44,53,58]};
var NS='http://www.w3.org/2000/svg',n=D.values.length,x0=20,yT=8,yB=225,H=265;
var svg=root.querySelector('svg'),wrap=root.querySelector('.fx-wrap'),tip=root.querySelector('.fx-tip');
var grid=[].slice.call(svg.querySelectorAll('.fx-grid')),labs=[].slice.call(svg.querySelectorAll('.fg-t-dim'));
var S={x1:620,step:0,pts:[],hits:[],cur:-1};
function el(tag,attrs,cls){var e=document.createElementNS(NS,tag);for(var k in attrs)e.setAttribute(k,attrs[k]);if(cls)e.setAttribute('class',cls);return e}
function num(v){return v===null||v===undefined||v===''||!isFinite(Number(v))?null:Number(v)}
function X(i){return x0+i*S.step}
function Y(v){return yB-v/D.max*(yB-yT)}
function curve(p){
var d='M'+p[0][0]+' '+p[0][1];
for(var i=0;i<p.length-1;i++){
var a=p[i-1]||p[i],b=p[i],c=p[i+1],e=p[i+2]||c;
var lo=Math.min(b[1],c[1]),hi=Math.max(b[1],c[1]);
var y1=Math.min(hi,Math.max(lo,b[1]+(c[1]-a[1])/6)),y2=Math.min(hi,Math.max(lo,c[1]-(e[1]-b[1])/6));
d+=' C'+(b[0]+(c[0]-a[0])/6)+' '+y1+' '+(c[0]-(e[0]-b[0])/6)+' '+y2+' '+c[0]+' '+c[1];
}
return d;
}
function hide(){
if(S.cur>=0&&S.pts[S.cur])S.pts[S.cur].classList.remove('is-on');
S.cur=-1;tip.classList.remove('is-on');
}
function show(i){
hide();
var v=num(D.values[i]);
if(v===null)return;
S.cur=i;S.pts[i].classList.add('is-on');
tip.textContent=D.name+' '+v+' '+D.unit;
tip.classList.add('is-on');
var tw=tip.getBoundingClientRect().width,th=tip.getBoundingClientRect().height,px=X(i),py=Y(v),top=py-12-th;
if(top<yT)top=py+12;
tip.style.left=Math.max(x0,Math.min(S.x1-tw,px-tw/2))+'px';
tip.style.top=top+'px';
}
function draw(){
hide();
var W=Math.min(640,svg.getBoundingClientRect().width||640);
S.x1=W-20;S.step=(S.x1-x0)/(n-1);
svg.setAttribute('viewBox','0 0 '+W+' '+H);
grid.forEach(function(g,k){var y=k===0?yT:k===1?(yT+yB)/2:yB;g.setAttribute('y1',y);g.setAttribute('y2',y);g.setAttribute('x2',S.x1)});
labs.forEach(function(t,j){t.setAttribute('x',X(j));t.setAttribute('y',yB+30);t.setAttribute('visibility',S.step<34&&j%2===1&&j!==n-1?'hidden':'visible')});
[].slice.call(svg.querySelectorAll('.fx-gen')).forEach(function(e){e.parentNode.removeChild(e)});
S.hits.forEach(function(h){wrap.removeChild(h)});
var runs=[],cur=[];
D.values.forEach(function(v,i){var y=num(v);if(y===null){if(cur.length)runs.push(cur);cur=[];return}cur.push([X(i),Y(y)])});
if(cur.length)runs.push(cur);
runs.forEach(function(p){
if(p.length<2)return;
var line=curve(p);
svg.appendChild(el('path',{d:line+' L'+p[p.length-1][0]+' '+yB+' L'+p[0][0]+' '+yB+' Z'},'fx-gen fx-area'));
svg.appendChild(el('path',{d:line},'fx-gen fx-line'));
});
S.pts=D.values.map(function(v,i){var y=num(v);if(y===null)return null;var c=el('circle',{cx:X(i),cy:Y(y),r:4},'fx-gen fx-pt');svg.appendChild(c);return c});
S.hits=[];
D.values.forEach(function(v,i){
if(num(v)===null)return;
var h=document.createElement('span');
h.className='fx-hit';h.tabIndex=0;h.setAttribute('data-i',i);
h.setAttribute('aria-label',D.name+' '+num(v)+' '+D.unit);
h.style.left=(X(i)-8)+'px';h.style.top=(Y(num(v))-8)+'px';
wrap.appendChild(h);S.hits.push(h);
});
}
wrap.addEventListener('mousemove',function(e){
var r=svg.getBoundingClientRect(),u=e.clientX-r.left,i=Math.max(0,Math.min(n-1,Math.round((u-x0)/S.step)));
show(i);
});
wrap.addEventListener('mouseleave',hide);
wrap.addEventListener('focusin',function(e){var i=e.target.getAttribute('data-i');if(i!==null)show(+i)});
wrap.addEventListener('focusout',hide);
draw();
window.addEventListener('resize',draw);
})();
</script>

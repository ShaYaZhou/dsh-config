# D04 环形占比

2 到 5 个类别、有意义的总量时用，总量放中心；要精确排名改用柱图。改数据只改脚本里 D 的 names、values、cls，值为 0 或缺失的类别会自动隐藏。改标题、总量、页脚在 HTML 中改，总量须等于各类之和。增减类别时同步改 D 与图例。焦点是第一扇区，悬停或 Tab 聚焦时外移并显示读数。示例内容只是占位，使用时整套换成用户的内容，包括标题、副标题、标签、页脚和一句话概述。

<script>
(function(){
var root=document.querySelector('.fg[data-fig="d04"]');
if(!root||root.getAttribute('data-ready'))return;
root.setAttribute('data-ready','1');
var D={unit:'次',names:['搜索','推荐','直接访问','邮件','其他'],cls:['fx-s1','fx-s2','fx-s3','fx-s4','fx-s0'],values:[920,610,480,240,150]};
var NS='http://www.w3.org/2000/svg',C=125,R=125,r0=77.5,total=0;
var wrap=root.querySelector('.fx-donut'),svg=root.querySelector('svg'),tip=root.querySelector('.fx-tip'),leg=root.querySelectorAll('.fx-li');
var S={secs:[],rings:[],mid:[],cur:-1};
function el(tag,attrs,cls){var e=document.createElementNS(NS,tag);for(var k in attrs)e.setAttribute(k,attrs[k]);if(cls)e.setAttribute('class',cls);return e}
function num(v){return v===null||v===undefined||v===''||!isFinite(Number(v))?null:Number(v)}
function pt(r,a){return (C+r*Math.sin(a)).toFixed(2)+' '+(C-r*Math.cos(a)).toFixed(2)}
function sec(ra,rb,a0,a1){var big=a1-a0>Math.PI?1:0;return 'M'+pt(rb,a0)+' A'+rb+' '+rb+' 0 '+big+' 1 '+pt(rb,a1)+' L'+pt(ra,a1)+' A'+ra+' '+ra+' 0 '+big+' 0 '+pt(ra,a0)+' Z'}
var items=[];
D.values.forEach(function(v,i){var x=num(v);if(x!==null&&x>0){items.push({i:i,v:x});total+=x}else if(leg[i])leg[i].style.display='none'});
var a=0;
items.forEach(function(it){
var a1=a+it.v/total*2*Math.PI;
it.a0=a;it.a1=a1;
S.mid[it.i]=(a+a1)/2;
var p=el('path',{d:sec(r0,R,a,a1),tabindex:'0','data-i':it.i},'fx-gen fx-sec '+D.cls[it.i]);
svg.appendChild(p);S.secs[it.i]=p;
a=a1;
});
items.forEach(function(it){
var g=el('path',{d:sec(r0-3,R+3,it.a0,it.a1)},'fx-gen fx-ring');
svg.appendChild(g);S.rings[it.i]=g;
});
function hide(){
if(S.cur>=0){S.secs[S.cur].style.transform='';S.rings[S.cur].style.transform='';S.rings[S.cur].classList.remove('is-on');S.cur=-1}
tip.classList.remove('is-on');
}
function show(i,focus){
hide();S.cur=i;
var m=S.mid[i],t='translate('+(4*Math.sin(m)).toFixed(2)+'px,'+(-4*Math.cos(m)).toFixed(2)+'px)';
S.secs[i].style.transform=t;
if(focus){S.rings[i].style.transform=t;S.rings[i].classList.add('is-on')}
tip.textContent=D.names[i]+' '+D.values[i]+' '+D.unit+' · '+(D.values[i]/total*100).toFixed(1)+'%';
tip.classList.add('is-on');
var sc=svg.getBoundingClientRect().width/250,tw=tip.offsetWidth,th=tip.offsetHeight;
var px=(C+(R+16)*Math.sin(m))*sc,py=(C-(R+16)*Math.cos(m))*sc;
tip.style.left=Math.max(0,Math.min(wrap.clientWidth-tw,px-tw/2))+'px';
tip.style.top=Math.max(0,Math.min(wrap.clientHeight-th,py-th/2))+'px';
}
wrap.addEventListener('mouseover',function(e){var i=e.target.getAttribute('data-i');if(i===null)hide();else show(+i,false)});
wrap.addEventListener('mouseleave',hide);
wrap.addEventListener('focusin',function(e){var i=e.target.getAttribute('data-i');if(i!==null)show(+i,true)});
wrap.addEventListener('focusout',hide);
})();
</script>

# I01 单控件步进

想一步一步看滑动窗口怎样移动、算式怎样变化时用，由读者用按钮控制节奏。替换数值与四步的窗口位置、算式；默认第 1 步，刷新后回到第 1 步，不保存状态。增减步数时只改脚本里的 steps 数组。示例内容只是占位，使用时整套换成用户的内容，包括标题、副标题、标签、页脚和一句话概述。

<script>
(function(){
var root=document.querySelector('.fg[data-fig="i01"]');
if(!root||root.getAttribute('data-ready'))return;
root.setAttribute('data-ready','1');
var ins=root.querySelectorAll('.fx-i01-ing .fg-cell');
var outs=root.querySelectorAll('.fx-i01-outg .fg-cell');
var eq=root.querySelector('.fx-i01-eq');
var steps=[[0,0,0,'1×1 + 2×2 + 3×3 + 1×4 = 18'],[0,1,1,'2×1 + 0×2 + 1×3 + 4×4 = 21'],[1,0,2,'3×1 + 1×2 + 0×3 + 2×4 = 13'],[1,1,3,'1×1 + 4×2 + 2×3 + 5×4 = 35']];
var cur=0,n=steps.length;
function show(){
var s=steps[cur];
for(var k=0;k<ins.length;k++){
var r=Math.floor(k/3)-s[0];
var c=k%3-s[1];
ins[k].classList.toggle('fx-i01-win',r>=0&&r<2&&c>=0&&c<2);
}
for(var o=0;o<outs.length;o++){outs[o].classList.toggle('is-key',o===s[2]);}
eq.textContent=s[3];
}
root.querySelector('[data-act="prev"]').addEventListener('click',function(){cur=(cur+n-1)%n;show();});
root.querySelector('[data-act="next"]').addEventListener('click',function(){cur=(cur+1)%n;show();});
})();
</script>
